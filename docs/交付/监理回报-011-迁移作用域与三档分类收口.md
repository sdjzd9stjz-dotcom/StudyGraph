### 2.1 实现（`claim-support.ts`）

```ts
const normalizedName = normalizeForCompare(name)
const normalizedDefinition = normalizeForCompare(definition)
const residual = normalizedDefinition.split(normalizedName).join('')

if (!normalizedDefinition.includes(normalizedName)) reasons.push('NAME_NOT_IN_DEFINITION')
if (normalizedDefinition === normalizedName) reasons.push('DEFINITION_EQUALS_NAME')   // 保留，别删
if (residual.length < MIN_RESIDUAL_CHARS) reasons.push('RESIDUAL_TOO_SHORT')
```

```ts
const hasHumanDecision = !!entity.reviewedBy

if (hasHumanDecision)        // 人工裁决过：只查 A1
```
